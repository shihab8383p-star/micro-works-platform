━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 1: CREATE GITHUB REPOSITORY (2 minutes)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Go to: https://github.com/new

2. Fill in:
   Repository name: microjob-platform
   Description: Micro Works - Microjob platform with 50 tasks
   Visibility: ✅ Public
   
3. DON'T CHECK:
   ❌ Add a README file
   ❌ Add .gitignore
   ❌ Choose a license

4. Click: "Create repository"

5. You'll see a page with commands - IGNORE THEM!

6. Copy YOUR repository URL:
   Example: https://github.com/YOUR_USERNAME/microjob-platform.git

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 2: PUSH YOUR CODE TO GITHUB (1 minute)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Open Command Prompt and run these commands:

cd "c:\Users\Shihab\Documents\Rockstar Games\Social Club\microjob-platform"

git remote add origin https://github.com/YOUR_USERNAME/microjob-platform.git

git branch -M main

git push -u origin main

When prompted for login:
  Username: YOUR_GITHUB_USERNAME
  Password: Use Personal Access Token (see below)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

HOW TO GET PERSONAL ACCESS TOKEN:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Go to: https://github.com/settings/tokens/new

2. Token name: Micro Works Deployment

3. Expiration: No expiration (or 90 days)

4. Select scopes: ✅ repo (Full control of private repositories)

5. Click: "Generate token"

6. COPY THE TOKEN (you won't see it again!)

7. Use this token as password when pushing

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 3: SETUP MONGODB ATLAS (3 minutes)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Go to: https://www.mongodb.com/cloud/atlas/register

2. Sign up with Google (easiest)

3. Choose: FREE tier (M0 Sandbox)

4. Provider: AWS

5. Region: ap-south-1 (Mumbai) - closest to Bangladesh

6. Cluster Name: MicroWorks

7. Click "Create"

8. Create Database User:
   - Username: microworks
   - Password: (Click "Autogenerate" and SAVE IT!)
   - Role: Atlas Admin
   - Click "Add User"

9. Add IP Whitelist:
   - Click "Network Access"
   - Click "Add IP Address"
   - Select "Allow Access from Anywhere"
   - Confirm

10. Get Connection String:
    - Click "Database" → "Connect"
    - Choose "Connect your application"
    - Copy the string
    - Replace <password> with your actual password
    - Add /microjob-platform before the ?
    
Example:
mongodb+srv://microworks:YOUR_PASSWORD@microworks.xxxxx.mongodb.net/microjob-platform?retryWrites=true&w=majority

SAVE THIS CONNECTION STRING!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 4: DEPLOY TO RENDER.COM (5 minutes)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Go to: https://render.com

2. Click: "Get Started for Free"

3. Sign up with GitHub

4. Authorize Render

5. Click: "New +" → "Web Service"

6. Connect your repository: microjob-platform

7. Configure:
   Name: micro-works
   Region: Singapore
   Branch: main
   Runtime: Node
   Build Command: npm install
   Start Command: npm start
   Plan: Free

8. Click "Advanced" and add Environment Variables:

   PORT = 5000
   NODE_ENV = production
   MONGODB_URI = [YOUR_CONNECTION_STRING_FROM_STEP_3]
   JWT_SECRET = microworks_jwt_secret_2024_CHANGE_THIS
   SITE_NAME = Micro Works
   MINIMUM_WITHDRAWAL = 500
   CURRENCY = BDT

9. Click "Create Web Service"

10. Wait 5-10 minutes for deployment

11. Your site will be live at: https://micro-works.onrender.com

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
